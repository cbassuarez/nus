1-- every new terminal gets its own hue; pages opened
2-- from it stay in the family, a touch lighter.
3function new_tab(ctx)
4  if ctx.parent then
5    return { bg = mix(ctx.parent.bg, "#ffffff", 0.06),
6             signal = ctx.parent.signal }
7  end
8  if ctx.kind == "terminal" then
9    local turn = (ctx.index - 1) * 0.11
10    local light = ctx.theme == "ink" and 0.11 or 0.93
11    return { bg = hsl(turn, 0.18, light), signal = hue(ctx.signal, turn) }
12  end
13end
14
15function new_space(ctx)
16  return { signal = hue("#c8102e", (ctx.index - 1) * 0.17) }
17end
18
19-- boosts: css and js a site gets, every visit
20function on_page(ctx)
21  if ctx.host == "github.com" then
22    return { css = "body{font-family:'IBM Plex Mono'}" }
23  end
24end

26-- sounds: which cue an event plays (nil = quiet)
27function on_event(ev)
28  if ev.name == "tab.close" then return "droplet" end
29end
